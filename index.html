How This Local Voice Memo Diarizer Works

Whisper transcribes words, but lacks speaker identities. WhisperX aligns transcript phonemes with Pyannote's neural voice embeddings to pinpoint exactly who said what and when, with sub-second accuracy.

diarize_service.py (Local FastAPI + WhisperX Backend)

Save this file on your Mac/PC or MyCloud attached machine. When running, this web interface will connect directly to it to process real M4A audio files on your local GPU/CPU. 

# Complete Local WhisperX & Pyannote Microservice
import os
import torch
import whisperx
from fastapi import FastAPI, UploadFile, File, Form
from fastapi.middleware.cors import CORSMiddleware
import shutil

app = FastAPI(title="Local Audio Diarization Service")

# Allow the single-page dashboard to communicate locally
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

DEVICE = "cuda" if torch.cuda.is_available() else ("mps" if torch.backends.mps.is_available() else "cpu")
BATCH_SIZE = 8 if DEVICE != "cpu" else 4
COMPUTE_TYPE = "float16" if DEVICE == "cuda" else "int8"

# 1. Load faster-whisper model
print(f"Loading Whisper model on {DEVICE}...")
whisper_model = whisperx.load_model("small", DEVICE, compute_type=COMPUTE_TYPE)

# 2. Pyannote Diarization
HF_TOKEN = os.getenv("HF_TOKEN", "")
diarize_model = whisperx.DiarizationPipeline(use_auth_token=HF_TOKEN, device=DEVICE)

@app.get("/health")
def health():
    return {"status": "online", "device": DEVICE, "compute_type": COMPUTE_TYPE}

@app.post("/process")
async def process_audio(file: UploadFile = File(...), min_speakers: int = Form(1), max_speakers: int = Form(5), hf_token: str = Form(None)):
    temp_path = f"/tmp/{file.filename}"
    with open(temp_path, "wb") as buffer:
        shutil.copyfileobj(file.file, buffer)

    # 1. Transcribe
    audio = whisperx.load_audio(temp_path)
    result = whisper_model.transcribe(audio, batch_size=BATCH_SIZE)

    # 2. Align timestamps to word phonemes
    align_model, metadata = whisperx.load_align_model(language_code=result["language"], device=DEVICE)
    result = whisperx.align(result["segments"], align_model, metadata, audio, DEVICE, return_char_alignments=False)

    # 3. Speaker Diarization
    diarize_segments = diarize_model(audio, min_speakers=min_speakers, max_speakers=max_speakers)
    diarized_result = whisperx.assign_word_speakers(diarize_segments, result)

    # Format cleanly for frontend
    segments = []
    for seg in diarized_result["segments"]:
        segments.append({
            "start": round(seg["start"], 2),
            "end": round(seg["end"], 2),
            "speaker": seg.get("speaker", "SPEAKER_00"),
            "text": seg["text"].strip()
        })

    os.remove(temp_path)
    return {"status": "success", "duration": audio.shape[0]/16000, "segments": segments}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8080)
